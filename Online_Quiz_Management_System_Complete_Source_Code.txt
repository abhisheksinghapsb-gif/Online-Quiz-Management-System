  16 |  */
  17 | public class MultipleChoiceQuestion extends Question {
  18 | 
  19 |     private final List<String> options;

  23 |                                   DifficultyLevel difficulty, List<String> options, int correctOptionIndex)
  24 |             throws InvalidQuestionException {

  26 | 
  27 |         if (options == null || options.size() < 2) {
  28 |             throw new InvalidQuestionException("MCQ must have at least 2 options! Found: " 
  29 |                     + (options == null ? 0 : options.size()));
  30 |         }
  31 |         if (correctOptionIndex < 0 || correctOptionIndex >= options.size()) {
  32 |             throw new InvalidQuestionException("Correct option index (" + correctOptionIndex 
  33 |                     + ") is out of valid range [0 to " + (options.size() - 1) + "]!");
  34 |         }
  35 | 
  36 |         this.options = new ArrayList<>(options);
  37 |         this.correctOptionIndex = correctOptionIndex;
  38 |     }

  40 |     public List<String> getOptions() {
  41 |         return Collections.unmodifiableList(options);

  46 |     }
  47 | 
  48 |     @Override
  49 |     public void displayQuestion() {
  50 |         displayHeader();
  51 |         char label = 'A';
  52 |         for (String option : options) {
  53 |             System.out.printf("   [%c] %s%n", label++, option);
  54 |         }
  55 |         System.out.println("   (Enter choice A-" + (char)('A' + options.size() - 1) + ")");

  57 | 
  58 |     @Override
  59 |     public boolean checkAnswer(String studentAnswer) throws InvalidOptionException {
  60 |         if (studentAnswer == null || studentAnswer.trim().isEmpty()) {
  61 |             throw new InvalidOptionException("Empty input", "A to " + (char)('A' + options.size() - 1));
  62 |         }
  63 |

  66 |             throw new InvalidOptionException(studentAnswer, "Single character A to " + (char)('A' + options.size() - 1));
  67 |         }

  70 |         int selectedIndex = choice - 'A';

  73 |             throw new InvalidOptionException(studentAnswer, "A to " + (char)('A' + options.size() - 1));
  74 |         }
  75 | 
  76 |         return selectedIndex == correctOptionIndex;
  77 |     }
  78 |

 367 |         public StaticFileHandler(Path rootDir) {
 368 |             this.rootDir = rootDir;
 369 |         }
 370 | 
 371 |         @Override
 372 |         public void handle(HttpExchange exchange) throws IOException {
 373 |             String pathStr = exchange.getRequestURI().getPath();
 374 |             if (pathStr == null || pathStr.equals("/") || pathStr.isEmpty()) {
 375 |                 pathStr = "/index.html";
 376 |             }
 377 | 
 378 |             // Prevent path traversal
 379 |             Path resolved = rootDir.resolve(pathStr.substring(1)).normalize();
 380 |             if (!resolved.startsWith(rootDir) || !Files.exists(resolved) || Files.isDirectory(resolved)) {
 381 |                 String notFound = "<h1>404 Not Found</h1><p>The requested file was not found.</p>";
 382 |                 exchange.sendResponseHeaders(404, notFound.length());
 383 |                 OutputStream os = exchange.getResponseBody();
 384 |                 os.write(notFound.getBytes(StandardCharsets.UTF_8));
 385 |                 os.close();
 386 |                 return;
 387 |             }
 388 | 
 389 |             String contentType = getMimeType(resolved.toString());
 390 |             exchange.getResponseHeaders().set("Content-Type", contentType);
 391 |             byte[] bytes = Files.readAllBytes(resolved);
 392 |             exchange.sendResponseHeaders(200, bytes.length);
 393 |             OutputStream os = exchange.getResponseBody();
 394 |             os.write(bytes);
 395 |             os.close();
 396 |         }

 440 |     private static String quote(String str) {
 441 |         if (str == null) return "\"\"";
 442 |         StringBuilder sb = new StringBuilder("\"");
 443 |         for (char c : str.toCharArray()) {
 444 |             switch (c) {
 445 |                 case '"': sb.append("\\\""); break;
 446 |                 case '\\': sb.append("\\\\"); break;
 447 |                 case '\b': sb.append("\\b"); break;
 448 |                 case '\f': sb.append("\\f"); break;
 449 |                 case '\n': sb.append("\\n"); break;
 450 |                 case '\r': sb.append("\\r"); break;
 451 |                 case '\t': sb.append("\\t"); break;
 452 |                 default:
 453 |                     if (c < ' ') {
 454 |                         sb.append(String.format("\\u%04x", (int) c));
 455 |                     } else {
 456 |                         sb.append(c);
 457 |                     }
 458 |             }
 459 |         }
 460 |         sb.append("\"");
 461 |         return sb.toString();
 462 |     }

 464 |     private static Map<String, String> parseSimpleJson(String json) {
 465 |         Map<String, String> map = new HashMap<>();
 466 |         if (json == null || json.trim().isEmpty()) return map;
 467 | 
 468 |         String trimmed = json.trim();
 469 |         if (trimmed.startsWith("{")) trimmed = trimmed.substring(1);
 470 |         if (trimmed.endsWith("}")) trimmed = trimmed.substring(0, trimmed.length() - 1);
 471 | 
 472 |         String[] pairs = trimmed.split(",(?=(?:[^\"]*\"[^\"]*\")*[^\"]*$)");
 473 |         for (String pair : pairs) {
 474 |             String[] kv = pair.split(":(?=(?:[^\"]*\"[^\"]*\")*[^\"]*$)", 2);
 475 |             if (kv.length == 2) {
 476 |                 String key = cleanQuotes(kv[0].trim());
 477 |                 String val = cleanQuotes(kv[1].trim());
 478 |                 map.put(key, val);
 479 |             }
 480 |         }
 481 |         return map;
 482 |     }

 484 |     private static Map<String, String> extractNestedMap(String json, String parentKey) {
 485 |         Map<String, String> result = new HashMap<>();
 486 |         int keyIndex = json.indexOf("\"" + parentKey + "\"");
 487 |         if (keyIndex == -1) return result;
 488 | 
 489 |         int openBrace = json.indexOf("{", keyIndex);
 490 |         if (openBrace == -1) return result;
 491 | 
 492 |         int closeBrace = json.indexOf("}", openBrace);
 493 |         if (closeBrace == -1) return result;
 494 | 
 495 |         String inner = json.substring(openBrace + 1, closeBrace);
 496 |         String[] pairs = inner.split(",");
 497 |         for (String p : pairs) {
 498 |             String[] kv = p.split(":");
 499 |             if (kv.length == 2) {
 500 |                 result.put(cleanQuotes(kv[0].trim()), cleanQuotes(kv[1].trim()));
 501 |             }
 502 |         }
 503 |         return result;
 504 |     }

 506 |     private static String cleanQuotes(String s) {
 507 |         if (s.startsWith("\"") && s.endsWith("\"") && s.length() >= 2) {
 508 |             return s.substring(1, s.length() - 1);
 509 |         }
 510 |         return s;
 511 |     }
